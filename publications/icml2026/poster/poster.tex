\begin{block}{The Task}
Given a \textbf{north-up overhead map} with labeled candidate directions and a
\textbf{Street View photo} taken at the same intersection, infer \emph{which
labeled direction the camera was facing}. The camera sits at the intersection
center; letters point outward along each road.
\vspace{0.4cm}
\begin{center}
\includegraphics[width=0.43\linewidth]{\FIG/sample-question-1-map.jpg}\hfill
\includegraphics[width=0.43\linewidth]{\FIG/sample-question-1-sv.jpg}
\end{center}
\vspace{0.2cm}
\centering\normalsize\textbf{Ground truth: D} (camera faces north).
\par\vspace{0.2cm}
\footnotesize\raggedright\emph{Solve by aligning stable map geometry (roads,
parcels, landmarks) with the ground-level view --- not transient cues.}
\end{block}

\begin{block}{Benchmark \& Data (open)}
\begin{itemize}
  \item \textbf{m2sv-20k} --- 20{,}000 real intersections (10k train / 10k
        validation) across \textbf{32 cities}, multiple continents, controlled
        candidate ambiguity.
  \item \textbf{m2sv-sft-11k} --- 11k curated structured reasoning traces for SFT.
  \item Released as \textbf{blueprints} (coordinates + azimuths) plus rendering
        code to respect map licensing; all eval/training code public.
\end{itemize}
\end{block}

\begin{block}{Human baseline \& agreement}
\begin{center}
\includegraphics[width=0.86\linewidth]{\FIG/human_accuracy_dist.png}
\end{center}
\footnotesize Attentive humans converge (median pairwise $\kappa=0.48$, up to
$0.76$); Gemini-3-Pro sits \emph{below} the human consensus (median $\kappa=0.42$),
Qwen3-VL-235B lower still ($0.15$). The gap shows up in \emph{agreement structure},
not only accuracy.
\end{block}

\begin{block}{Difficulty: the hardest-tier collapse}
\begin{center}
\includegraphics[width=0.80\linewidth]{\FIG/human_difficulty_decircular.png}
\end{center}
\footnotesize Difficulty bins from \textbf{human response time}; accuracy from a
model-independent axis. Humans degrade gently ($83\!\to\!63\%$) and stay above both
models; models lose the most ground on the hardest tier, falling to (Qwen) or
toward (Gemini, $50\%$) chance. \textbf{Symmetric} intersections (azimuth-gap ratio
$S=\frac{\max}{\min}$) are hardest for all models.
\end{block}

\begin{block}{Takeaways}
\begin{itemize}
  \item m2sv isolates a scalable \textbf{map$\leftrightarrow$street-view alignment}
        primitive frontier VLMs have \emph{not} solved (best 65\% vs human 72/95\%).
  \item Humans beat the best model \emph{and} agree with one another; models lie
        outside that consensus and collapse on the hardest, symmetric cases.
  \item SFT$+$RL on m2sv helps ($+$10 pts) but transfer is limited --- failures are
        about \emph{adapting reasoning to ambiguity}, not missing data.
\end{itemize}
\end{block}
