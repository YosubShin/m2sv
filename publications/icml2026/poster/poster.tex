\begin{block}{The Task}
Given a \textbf{north-up overhead map} with labeled candidate directions and a
\textbf{Street View photo} taken at the same intersection, infer \emph{which
labeled direction the camera was facing}. The camera sits at the intersection
center; letters point outward along each road.
\vspace{0.6cm}
\begin{center}
\includegraphics[width=0.46\linewidth]{\FIG/sample-question-1-map.jpg}\hfill
\includegraphics[width=0.46\linewidth]{\FIG/sample-question-1-sv.jpg}
\end{center}
\vspace{0.3cm}
\footnotesize\emph{Solve by aligning stable map geometry (roads, parcels,
landmarks) with the ground-level view --- not transient cues.}
\end{block}

\begin{block}{Why it is hard}
\begin{itemize}
  \item \textbf{Allocentric $\leftrightarrow$ egocentric} alignment across two
        very different viewpoints.
  \item \textbf{Temporal mismatch}: map and photo may be captured years apart.
  \item \textbf{Geometric ambiguity}: symmetric intersections admit multiple
        plausible headings and demand fine-grained cue escalation.
\end{itemize}
\end{block}

\begin{block}{Benchmark \& Data (open)}
\begin{itemize}
  \item \textbf{m2sv-20k} --- 20{,}000 real intersections (10k train / 10k
        validation) across \textbf{32 cities}, multiple continents, with
        controlled candidate ambiguity.
  \item \textbf{m2sv-sft-11k} --- 11k curated structured reasoning traces for
        supervised fine-tuning.
  \item Released as \textbf{blueprints} (coordinates + azimuths) plus rendering
        code to respect map licensing; all eval/training code public.
\end{itemize}
\end{block}

\end{column}

\begin{column}{\colw}

\begin{block}{Difficulty: the gap widens where it matters}
\begin{center}
\includegraphics[width=0.99\linewidth]{\FIG/human_difficulty_decircular.png}
\end{center}
\footnotesize Difficulty bins from \textbf{human response time}; accuracy from
\emph{separate} annotators (de-circularized). Easy$\to$Hard: humans
$83\!\to\!63\%$, Gemini $82\!\to\!50\%$, Qwen $42\!\to\!29\%$. Gemini nearly
matches humans on easy items but trails by \textbf{13 pts} on hard ones.
\vspace{0.4cm}

\textbf{Road-azimuth symmetry.} We define a structural difficulty metric
$S=\dfrac{\max\text{ angular gap}}{\min\text{ angular gap}}$. Stratifying
3-option items by $S$, \emph{symmetric} intersections are hardest for all
models.
\end{block}

\begin{block}{How models fail}
\begin{itemize}
  \item \textbf{Ego$\leftrightarrow$allo confusion} --- left/right inversions at
        4-way junctions.
  \item \textbf{Unreliable cues} --- roof color, shadows, blur drive confident
        wrong eliminations.
  \item \textbf{Landmark misbinding / false projection} --- attaching distant or
        absent landmarks to the intersection.
  \item \textbf{Symmetry traps} --- fail to escalate to lane counts, markings,
        curb geometry that resolve ambiguity.
\end{itemize}
\vspace{0.2cm}
\footnotesize Common thread: models apply \emph{uniform} reasoning across
difficulty; humans escalate effort on ambiguous cases.
\end{block}

\begin{block}{Takeaways}
\begin{itemize}
  \item m2sv isolates a core, scalable \textbf{map$\leftrightarrow$street-view
        alignment} primitive that frontier VLMs have \emph{not} solved.
  \item Humans beat the best model and \textbf{agree with each other}; models
        lie outside that consensus.
  \item Difficulty is real and structural --- failures concentrate on
        \emph{symmetric} geometry.
\end{itemize}
